local function EnviarNotificacao(titulo, texto, duracao)
    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = titulo, Text = texto, Duration = duracao or 2
        })
    end)
end

local function AtualizarChar(nc)
    Char = nc or LP.Character or LP.CharacterAdded:Wait()
    Root = Char:WaitForChild("HumanoidRootPart", 5)
    Hum = Char:WaitForChild("Humanoid", 5)
end

local function Soco()
    pcall(function()
        VirtualInputManager:SendMouseButtonEvent(0, 0, 0, true, game, 0)
        task.wait(0.08)
        VirtualInputManager:SendMouseButtonEvent(0, 0, 0, false, game, 0)
    end)
end

local function verificarSeEstouVivo()
    if not Hum then return false end
    return Hum.Health > 0
end

-- ========== BYPASS (ÚNICO, COM TRAVA) ==========
local function ExecuteBypass()
    if fazendoBypass then return false end
    fazendoBypass = true
    
    StatusAutoFarm = "EXECUTANDO BYPASS"
    
    if not Char or not Root then AtualizarChar() end
    
    local sucesso = false
    
    if Root then
        pcall(function()
            local jogadores = Players:GetPlayers()
            for i = #jogadores, 1, -1 do 
                if jogadores[i] == LP then table.remove(jogadores, i) end 
            end
            
            if #jogadores > 0 then
                local alvo = jogadores[math.random(1, #jogadores)]
                if alvo and alvo.Character and alvo.Character:FindFirstChild("HumanoidRootPart") then
                    local pos = alvo.Character.HumanoidRootPart.Position + Vector3.new(0, 3, 0)
                    Root.CFrame = CFrame.new(pos)
                    task.wait(0.5)
                    if Root and Root.Parent and (Root.Position - pos).Magnitude < 25 then
                        sucesso = true
                    end
                end
            end
        end)
    end
    
    fazendoBypass = false
    return sucesso
end

-- ========== ROTA DO VOID ==========
local function rotaDoVoid()
    pcall(function()
        if not Root or not Hum then return end
        Hum.PlatformStand = true
        local bv = Instance.new("BodyVelocity")
        bv.MaxForce = Vector3.new(9999999, 9999999, 9999999)
        bv.Velocity = Vector3.new(0, 100, 0)
        bv.Parent = Root
        task.wait(2)
        bv.Velocity = Vector3.new(-200, 0, 0)
        task.wait(3.6)
        bv.Velocity = Vector3.zero
        Hum.PlatformStand = false
        bv:Destroy()
    end)
end

-- ========== FAZER BYPASS COM LOOP CONTROLADO ==========
local function fazerBypassAteSucesso()
    local sucesso = false
    local tentativas = 0
    
    while AutoFarmAtivo and not sucesso do
        tentativas += 1
        print("💀 Bypass tentativa #" .. tentativas)
        
        sucesso = ExecuteBypass()
        
        if sucesso then
            StatusAutoFarm = "BYPASS CONCLUÍDO"
            print("✅ Bypass OK!")
        else
            print("❌ Falhou #" .. tentativas)
            rotaDoVoid()
            task.wait(5)
            if LP.Character then AtualizarChar(LP.Character) end
            task.wait(2)
        end
    end
    
    return sucesso
end

-- ========== ATACAR ALVO (TP + 1 SOCO + 2s) ==========
local function atacarAlvo(nomeAlvo)
    if trocandoAlvo then return true end
    trocandoAlvo = true
    
    local player = Players:FindFirstChild(nomeAlvo)
    if not player then
        StatusAutoFarm = nomeAlvo .. " SAIU"
        trocandoAlvo = false
        return true
    end
    
    if not player.Character then
        StatusAutoFarm = nomeAlvo .. " AGUARDANDO"
        trocandoAlvo = false
        return true
    end
    
    local targetRoot = player.Character:FindFirstChild("HumanoidRootPart")
    local targetHum = player.Character:FindFirstChildOfClass("Humanoid")
    
    if not targetRoot or not targetHum then
        StatusAutoFarm = nomeAlvo .. " SEM PARTES"
        trocandoAlvo = false
        return true
    end
    
    -- Não ignora só porque Health é 0 (pode ter corpo)
    if not Root then
        AtualizarChar()
    end
    
    if not Root then
        trocandoAlvo = false
        return false
    end
    
    -- TELEPORTA UMA VEZ (sem oscilar)
    local posicaoAtras = targetRoot.CFrame * CFrame.new(0, 0, -3)
    Root.CFrame = CFrame.lookAt(posicaoAtras.Position, targetRoot.Position)
    
    StatusAutoFarm = "TP PARA: " .. nomeAlvo
    task.wait(0.3)
    
    -- SOCO
    Soco()
    StatusAutoFarm = "SOCANDO: " .. nomeAlvo
    task.wait(0.2)
    
    -- ESPERA 2s
    StatusAutoFarm = "AGUARDANDO 2s: " .. nomeAlvo
    task.wait(2)
    
    trocandoAlvo = false
    return true
end

-- ========== PEGAR PRÓXIMO ALVO ==========
local function GetNextTarget()
    local nomes = {}
    for nome, _ in pairs(AlvosSelecionados) do
        local player = Players:FindFirstChild(nome)
        if player then
            table.insert(nomes, nome)
        end
    end
    
    if #nomes == 0 then return nil end
    
    table.sort(nomes)
    
    if AlvoIndex > #nomes then AlvoIndex = 1 end
    local alvo = nomes[AlvoIndex]
    AlvoIndex += 1
    if AlvoIndex > #nomes then AlvoIndex = 1 end
    
    return alvo
end

-- ========== CONTROLADOR PRINCIPAL (ÚNICO) ==========
local function StartAutoFarm()
    AutoFarmAtivo = true
    
    farmLoop = task.spawn(function()
        -- BYPASS ÚNICO
        local sucesso = fazerBypassAteSucesso()
        
        if not sucesso or not AutoFarmAtivo then
            StopAutoFarm()
            return
        end
        
        task.wait(1)
        
        -- LOOP PRINCIPAL
        while AutoFarmAtivo do
            -- VERIFICA MORTE
            if not verificarSeEstouVivo() then
                StatusAutoFarm = "MORTO - AGUARDANDO RESPAWN"
                
                -- ESPERA RESPAWN
                repeat 
                    task.wait(1) 
                until LP.Character and LP.Character:FindFirstChildOfClass("Humanoid") and LP.Character:FindFirstChildOfClass("Humanoid").Health > 0
                
                AtualizarChar()
                
                -- BYPASS DE NOVO
                StatusAutoFarm = "REFAZENDO BYPASS"
                local ok = fazerBypassAteSucesso()
                
                if not ok or not AutoFarmAtivo then
                    StopAutoFarm()
                    return
                end
                
                StatusAutoFarm = "BYPASS CONCLUÍDO"
                task.wait(1)
            end
            
            -- PEGA PRÓXIMO ALVO
            local alvo = GetNextTarget()
            if not alvo then
                StatusAutoFarm = "SEM ALVOS - AGUARDANDO"
                task.wait(2)
                continue
            end
            
            -- ATACA O ALVO
            atacarAlvo(alvo)
        end
    end)
end

local function StopAutoFarm()
    AutoFarmAtivo = false
    AlvoIndex = 1
    trocandoAlvo = false
    fazendoBypass = false
    
    if farmLoop then
        task.cancel(farmLoop)
        farmLoop = nil
    end
    
    StatusAutoFarm = "DESATIVADO"
end

local function RefreshPlayerList()
    for _, child in pairs(PlayerList:GetChildren()) do
        if child:IsA("Frame") then child:Destroy() end
    end
    
    local nomes = {}
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LP then table.insert(nomes, p.Name) end
    end
    table.sort(nomes)
    
    PlayerList.CanvasSize = UDim2.new(0, 0, 0, #nomes * 34)
    
    for _, nome in pairs(nomes) do
        local Row = Instance.new("Frame")
        Row.Parent = PlayerList
        Row.BackgroundColor3 = Color3.fromRGB(18, 38, 25)
        Row.BackgroundTransparency = 0.4
        Row.Size = UDim2.new(1, 0, 0, 30)
        Instance.new("UICorner", Row).CornerRadius = UDim.new(0, 6)
        
        local Check = Instance.new("TextButton")
        Check.Parent = Row
        Check.BackgroundTransparency = 1
        Check.Position = UDim2.new(0, 5, 0, 0)
        Check.Size = UDim2.new(0, 25, 1, 0)
        Check.Font = Enum.Font.GothamBold
        Check.Text = AlvosSelecionados[nome] and "☑" or "☐"
        Check.TextColor3 = AlvosSelecionados[nome] and Color3.fromRGB(0, 255, 128) or Color3.fromRGB(255, 255, 255)
        Check.TextSize = 16
        
        local NomeLabel = Instance.new("TextLabel")
        NomeLabel.Parent = Row
        NomeLabel.BackgroundTransparency = 1
        NomeLabel.Position = UDim2.new(0, 30, 0, 0)
        NomeLabel.Size = UDim2.new(1, -35, 1, 0)
        NomeLabel.Font = Enum.Font.Gotham
        NomeLabel.Text = nome
        NomeLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
        NomeLabel.TextSize = 12
        NomeLabel.TextXAlignment = Enum.TextXAlignment.Left
        
        Check.MouseButton1Click:Connect(function()
            if AlvosSelecionados[nome] then
                AlvosSelecionados[nome] = nil
                Check.Text = "☐"
                Check.TextColor3 = Color3.fromRGB(255, 255, 255)
            else
                AlvosSelecionados[nome] = true
                Check.Text = "☑"
                Check.TextColor3 = Color3.fromRGB(0, 255, 128)
            end
            
            local count = 0
            for _ in pairs(AlvosSelecionados) do count += 1 end
            Contador.Text = "Selecionados: " .. count
        end)
    end
    
    local count = 0
    for _ in pairs(AlvosSelecionados) do count += 1 end
    Contador.Text = "Selecionados: " .. count
end
